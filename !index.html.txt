    // HyperEVM Mainnet आधिकारिक सेटिंग्स
    const HYPER_RPC = "https://hyperliquid.xyz";
    const CHAIN_ID = 999;

    const INTEGRATOR_ID = "HyperTradeOpen";

    const provider = new ethers.JsonRpcProvider(HYPER_RPC);
    let signer = null;
    let userAddress = "";

    function toggleCustomToken() {
        const selectVal = document.getElementById("buy-token").value;
        const customBox = document.getElementById("custom-token-box");
        if (selectVal === "CUSTOM") {
            customBox.classList.remove("hidden");
        } else {
            customBox.classList.add("hidden");
        }
    }

    // 1. नया नॉन-कस्टोडियल वॉलेट बनाना
    async function createWallet() {
        const wallet = ethers.Wallet.createRandom().connect(provider);
        signer = wallet;
        userAddress = wallet.address;
        alert("⚠️ सुरक्षा चेतावनी: अपना 12 शब्दों का रिकवरी सीड फ्रेज़ सुरक्षित लिख लें। यह केवल आपके डिवाइस पर है:\n\n" + wallet.mnemonic.phrase);
        loadDashboard();
    }

    // 2. MetaMask / Trust Wallet को HyperEVM Mainnet से जोड़ना
    async function connectBrowserWallet() {
        if (!window.ethereum) {
            alert("कृपया ब्राउज़र में MetaMask या Trust Wallet एक्सटेंशन इंस्टॉल करें, या '+ नया HyperEVM वॉलेट बनाएं' चुनें।");
            return;
        }
        try {
            const browserProvider = new ethers.BrowserProvider(window.ethereum);
            try {
                await window.ethereum.request({
                    method: 'wallet_switchEthereumChain',
                    params: [{ chainId: CHAIN_HEX }],
                });
            } catch (switchError) {
                if (switchError.code === 4902) {
                    await window.ethereum.request({
                        method: 'wallet_addEthereumChain',
                        params: [{
                            chainId: CHAIN_HEX,
                            chainName: 'HyperEVM',
                            nativeCurrency: { name: 'HYPE', symbol: 'HYPE', decimals: 18 },
                            rpcUrls: [HYPER_RPC],
                            blockExplorerUrls: ['https://hyperevmscan.io']
                        }],
                    });
                }
            }
            signer = await browserProvider.getSigner();
            userAddress = await signer.getAddress();
            loadDashboard();
        } catch (err) {
            alert("वॉलेट कनेक्ट करने में त्रुटि: " + err.message);
        }
    }

    async function loadDashboard() {
        document.getElementById("auth-section").classList.add("hidden");
        document.getElementById("app-section").classList.remove("hidden");
        document.getElementById("user-address").innerText = userAddress;
        document.getElementById("explorer-link").href = `https://hyperevmscan.ioaddress/${userAddress}`;
        refreshBalance();
    }

    async function refreshBalance() {
        if (!userAddress) return;
        const bal = await provider.getBalance(userAddress);
        document.getElementById("hype-balance").innerText = parseFloat(ethers.formatEther(bal)).toFixed(4) + " HYPE";
    }
